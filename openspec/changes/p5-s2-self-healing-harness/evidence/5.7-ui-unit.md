### splitToolError (MessageBubble.tsx)

```ts
export function splitToolError(raw: string): SplitToolErrorResult {
  let parsed: unknown;
  try { parsed = JSON.parse(raw); }
  catch { return { body: raw, hint: null, examples: null }; }
  if (!parsed || typeof parsed !== "object") {
    return { body: JSON.stringify(parsed, null, 2), hint: null, examples: null };
  }
  const obj = parsed as Record<string, unknown>;
  let hint: string | null = null;
  if (typeof obj.hint === "string" && obj.hint.trim().length > 0) hint = obj.hint;
  let examples: unknown[] | null = null;
  if (Array.isArray(obj.examples)) examples = obj.examples;
  return { body: JSON.stringify(parsed, null, 2), hint, examples };
}
```

### AutoResumeBanner 纯函数 + 组件

```ts
export function buildAutoResumeBannerText(attempts: number, max: number): string {
  return `🔄 agent 自愈中... (尝试 ${attempts}/${max})`;
}
export function shouldShowAutoResumeBanner(attempts: number, inflight: boolean): boolean {
  return attempts > 0 && inflight;
}
```

```tsx
export function AutoResumeBanner({ sessionId, maxAttempts = 2 }: AutoResumeBannerProps) {
  const sid = useSessionsStore((s) => sessionId ?? s.active_sid);
  const session = useSessionsStore((s) => s.sessions[sid]);
  const attempts = session?.auto_resume_attempts ?? 0;
  const inflight = session?.inflight ?? false;
  if (!shouldShowAutoResumeBanner(attempts, inflight)) return null;
  return (
    <div role="status" data-testid="auto-resume-banner" style={{...}}>
      {buildAutoResumeBannerText(attempts, maxAttempts)}
    </div>
  );
}
```

### ws.ts 三个新 case

```ts
case "auto_resume_started": {
  const p = msg.payload || {};
  const target = p.session_id || sid;
  const attempt = typeof p.attempt === "number" ? p.attempt : 1;
  store.ensure(target);
  store.upsert(target, {
    auto_resume_attempts: attempt,
    inflight: true,
    status: "running",
  });
  break;
}
case "auto_resume_succeeded": {
  const target = (msg.payload || {}).session_id || sid;
  store.upsert(target, { auto_resume_attempts: 0 });
  break;
}
case "auto_resume_exhausted": {
  const p = msg.payload || {};
  const target = p.session_id || sid;
  const final_error = String(p.final_error ?? "auto-resume exhausted");
  const attempts_n = typeof p.attempts === "number" ? p.attempts : 0;
  store.upsert(target, { auto_resume_attempts: 0, status: "error", inflight: false });
  store.push_message(target, {
    role: "error",
    text: `自愈失败（${attempts_n} 次尝试）: ${final_error}`,
  });
  break;
}
```
